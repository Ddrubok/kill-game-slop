import assert from 'node:assert/strict';
import path from 'node:path';
import { withScene } from '../tools/scene.mjs';

const web = path.resolve('web');
await withScene(web, {}, async ({ page, errors }) => {
  await page.evaluate(() => __case.open('rapid'));
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(el => getComputedStyle(el).animationName), 'promotionModalIn');
  await page.evaluate(() => {
    const buttons = [...document.querySelectorAll('.promotion-piece-card')];
    buttons[0].click();
    buttons[1].click();
    buttons[0].click();
  });
  assert.deepEqual(await page.locator('.promotion-piece-card').evaluateAll(buttons => buttons.map(button => ({
    pressed: button.getAttribute('aria-pressed'),
    disabled: button.disabled,
    selected: button.classList.contains('promotion-selected'),
  }))), [
    { pressed: 'true', disabled: true, selected: true },
    ...Array(3).fill({ pressed: 'false', disabled: true, selected: false }),
  ]);
  await page.waitForTimeout(90);
  assert.deepEqual(await page.evaluate(() => __case.calls), []);
  await page.waitForTimeout(70);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{ owner: 'rapid', type: 'Queen' }]);

  await page.evaluate(() => {
    __case.open('old');
    const oldButton = document.querySelector('.promotion-piece-card');
    document.querySelector('.promotion-piece-card').click();
    __case.open('new');
    oldButton.click();
    document.querySelectorAll('.promotion-piece-card')[2].click();
  });
  await page.waitForTimeout(160);
  assert.deepEqual(await page.evaluate(() => __case.calls), [
    { owner: 'rapid', type: 'Queen' }, { owner: 'new', type: 'Bishop' },
  ]);

  await page.evaluate(() => {
    __case.open('removed');
    document.querySelector('.promotion-piece-card').click();
    document.querySelector('.promotion-modal-overlay').remove();
  });
  await page.waitForTimeout(160);
  assert.equal(await page.evaluate(() => __case.calls.length), 2);

  await page.evaluate(() => { __case.motion(true); __case.open('reduced'); });
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(el => getComputedStyle(el).animationName), 'none');
  assert.equal(await page.locator('.promotion-piece-card').first().evaluate(el => getComputedStyle(el).transitionDuration), '0s');
  await page.locator('.promotion-piece-card').nth(3).click();
  await page.screenshot({ path: path.resolve('work/promotion-reduced-pending.png') });
  await page.waitForTimeout(160);
  assert.deepEqual(await page.evaluate(() => __case.calls.at(-1)), { owner: 'reduced', type: 'Knight' });
  assert.deepEqual(errors, []);
});

await withScene(web, { reducedMotion: 'reduce' }, async ({ page, errors }) => {
  await page.evaluate(() => __case.open('system'));
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(el => getComputedStyle(el).animationName), 'none');
  assert.deepEqual(errors, []);
});

console.log('promotion repair checks passed');
