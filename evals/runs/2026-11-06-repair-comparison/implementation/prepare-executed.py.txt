from pathlib import Path
import hashlib, json, shutil, subprocess

ROOT = Path(__file__).resolve().parents[2]
PRIVATE = ROOT / '.orca/repair-comparison-20261106'
PUBLIC = ROOT / 'evals/runs/2026-11-06-repair-comparison'
SOURCE = ROOT / '.orca/validation/chess-full/web'

def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def write(p, value):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(value, ensure_ascii=False, indent=2)+'\n', encoding='utf-8')
def junction(p, target):
    subprocess.run(['powershell','-NoProfile','-Command',f"New-Item -ItemType Junction -Path '{p}' -Target '{target}' | Out-Null"], check=True)

assert not PRIVATE.exists(), 'Use a fresh experiment directory'
PRIVATE.mkdir(parents=True)
PUBLIC.mkdir(parents=True, exist_ok=True)

shutil.copytree(SOURCE, base, ignore=shutil.ignore_patterns('node_modules','.git','.orca','dist','.vite'))
directory=(base.parent.relative_to(ROOT)).as_posix()
subprocess.run(['git','apply','--reverse','--directory='+directory,'evals/real/promotion-2026-11-05/repair.patch'],cwd=ROOT,check=True)
expected=json.loads((ROOT/'.orca/validation/chess-full-source.json').read_text(encoding='utf-8'))['files']
assert {p.relative_to(base).as_posix():sha(p) for p in base.rglob('*') if p.is_file()}==expected
junction(base/'node_modules', SOURCE/'node_modules')
write(PUBLIC/'game-input.json', {'repository':'Ddrubok/ChessAlkkkaegi','commit':'974cb39bcc716815aee79117f4f710b367454050','files':expected})
