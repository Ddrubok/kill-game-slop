import assert from 'node:assert/strict';
import path from 'node:path';
import { writeFile } from 'node:fs/promises';
import { withScene } from '../tools/scene.mjs';

await withScene(path.resolve('web'), { viewport: { width: 360, height: 800 } }, async ({ page, errors }) => {
  await page.evaluate(() => __case.open('first', 'white'));
  assert.equal(await page.locator('[role="dialog"][aria-modal="true"]').count(), 1);
  assert.equal(await page.locator('.promotion-piece-card').count(), 4);
  await page.locator('[data-piece-type="Queen"]').click();
  assert.equal(await page.locator('[data-piece-type="Queen"]').getAttribute('aria-pressed'), 'true');
  assert.equal(await page.locator('.promotion-piece-card[aria-disabled="true"]').count(), 4);
  assert.equal(await page.locator('[data-piece-type="Queen"]').evaluate(button => document.activeElement === button), true);
  assert.equal(await page.locator('.promotion-piece-card.promotion-selected').count(), 1);
  assert.deepEqual(await page.evaluate(() => __case.calls), []);
  await writeFile(path.resolve('work/promotion-pending.png'), await page.screenshot());
  await page.locator('[data-piece-type="Rook"]').evaluate(button => button.click());
  await page.waitForTimeout(150);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'first', type:'Queen'}]);

  await page.evaluate(() => { __case.open('old'); document.querySelector('[data-piece-type="Bishop"]').click(); __case.open('new', 'black'); });
  await page.waitForTimeout(150);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'first', type:'Queen'}]);
  assert.equal(await page.locator('.promotion-piece-card[aria-disabled="true"]').count(), 0);
  await page.locator('[data-piece-type="Knight"]').click();
  await page.waitForTimeout(150);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'first', type:'Queen'}, {owner:'new', type:'Knight'}]);

  await page.evaluate(() => { __case.open('removed'); document.querySelector('[data-piece-type="Rook"]').click(); document.querySelector('.promotion-modal-overlay').remove(); });
  await page.waitForTimeout(150);
  assert.equal((await page.evaluate(() => __case.calls)).length, 2);

  await page.evaluate(() => { __case.motion(true); __case.open('reduced'); });
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(element => getComputedStyle(element).animationName), 'none');
  assert.equal(await page.locator('.promotion-piece-card').first().evaluate(element => getComputedStyle(element).transitionDuration), '0s');
  await page.locator('[data-piece-type="Rook"]').click();
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(element => getComputedStyle(element).animationName), 'none');
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'first', type:'Queen'}, {owner:'new', type:'Knight'}]);
  await page.waitForTimeout(150);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'first', type:'Queen'}, {owner:'new', type:'Knight'}, {owner:'reduced', type:'Rook'}]);
  assert.deepEqual(errors, []);
  console.log('promotion interaction checks passed');
});

await withScene(path.resolve('web'), { reducedMotion: 'reduce' }, async ({ page, errors }) => {
  await page.evaluate(() => __case.open('system-reduced'));
  assert.equal(await page.locator('.promotion-modal-overlay').evaluate(element => getComputedStyle(element).animationName), 'none');
  await page.locator('[data-piece-type="Bishop"]').focus();
  await page.keyboard.press('Enter');
  assert.equal(await page.locator('[data-piece-type="Bishop"]').getAttribute('aria-pressed'), 'true');
  await page.waitForTimeout(150);
  assert.deepEqual(await page.evaluate(() => __case.calls), [{owner:'system-reduced', type:'Bishop'}]);
  assert.deepEqual(errors, []);
  console.log('system motion and keyboard checks passed');
});
